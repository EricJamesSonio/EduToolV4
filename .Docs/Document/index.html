01 main.ts — Application Bootstrap
Entry point. Creates the NestJS application, wires global middleware, pipes, filters, interceptors, then starts listening. Everything that applies to ALL routes is registered here.
src/main.ts bootstrap 
async function bootstrap() {
  // 1. Create the NestJS app from AppModule (root module)
  const app = await NestFactory.create(AppModule, {
    logger: WinstonModule.createLogger(winstonConfig), // custom logger
  })

  // 2. Apply global prefix (all routes: /api/v1/...)
  app.setGlobalPrefix('api/v1')

  // 3. Apply global middleware (runs before guards/pipes/interceptors)
  app.use(RequestIdMiddleware())   // stamps x-request-id on every request
  app.use(helmet())               // security headers
  app.use(compression())          // gzip response bodies
  app.use(cookieParser())         // parse cookies (refresh token)
  app.enableCors({ origin: config.ALLOWED_ORIGINS, credentials: true })

  // 4. Global validation pipe — applies to ALL incoming DTOs
  //    whitelist strips unknown props, transform auto-converts types
  app.useGlobalPipes(
    new ValidationPipe({ whitelist: true, transform: true, forbidNonWhitelisted: true })
  )

  // 5. Global exception filters (outermost catch)
  app.useGlobalFilters(
    new AllExceptionFilter(logger),     // catch-all (unexpected errors)
    new HttpExceptionFilter(logger),    // HTTP errors (400, 401, 404...)
  )

  // 6. Global interceptors (wrap every controller method)
  app.useGlobalInterceptors(
    new LoggingInterceptor(logger),     // log req/res + duration
    new ResponseInterceptor(),           // wrap all responses in { data, meta }
  )

  // 7. Swagger setup (dev/staging only)
  if (config.NODE_ENV !== 'production') setupSwagger(app)

  // 8. Prisma graceful shutdown hook
  const prisma = app.get(DatabaseService)
  await prisma.enableShutdownHooks(app)

  await app.listen(config.PORT ?? 3000)
  logger.log(`🚀 EduTool API running on port ${config.PORT}`)
}

  async enableShutdownHooks(app: INestApplication) {
    // Prisma signals 'beforeExit' → call app.close() → NestJS lifecycle hooks run

06 core/middleware/request-id.middleware.ts
Runs before everything else. Stamps a unique UUID on every incoming request. This ID is threaded through all logs so you can trace a full request lifecycle across services.
src/core/middleware/request-id.middleware.ts middleware 
@Injectable()
export class RequestIdMiddleware implements NestMiddleware {

  use(req: Request, res: Response, next: NextFunction) {
    // Use incoming x-request-id if trusted proxy provides it, else generate one
    const requestId = req.headers['x-request-id'] ?? uuidv4()

    req['requestId']              = requestId         // attach to req object
    res.setHeader('x-request-id', requestId)  // echo back to client

    // AsyncLocalStorage — makes requestId available in any nested service call
    // without having to pass it through every function signature
    asyncLocalStorage.run({ requestId }, () => next())
  }
}

// Applied in AppModule.configure():
// consumer.apply(RequestIdMiddleware).forRoutes('*')
07 commons/guards/ — Auth & Roles Guards
Guards run after middleware but before interceptors and route handlers. They answer: "Is this request allowed to proceed?" Two guards stack: AuthGuard validates the JWT, RolesGuard checks the role against the route's requirements.
auth.guard.ts — JWT validation
src/commons/guards/auth.guard.ts guard 
@Injectable()
export class AuthGuard implements CanActivate {

  canActivate(ctx: ExecutionContext): boolean {
    const req = ctx.switchToHttp().getRequest()

    // 1. Extract Bearer token from Authorization header
    const token = this.extractToken(req.headers.authorization)
    if (!token) throw new UnauthorizedException('No token provided')

    try {
      // 2. Verify signature + expiry using JWT_ACCESS_SECRET
      const payload = this.jwt.verify(token, { secret: jwtConfig.accessSecret })

      // 3. Attach decoded payload to req so @CurrentUser() can read it
      // payload shape: { sub: userId, role: 'ADMIN'|'EDUCATOR'|'STUDENT', orgId }
      req.user = payload
      return true
    } catch {
      throw new UnauthorizedException('Invalid or expired token')
    }
  }

  private extractToken(header?: string) {
    // 'Bearer eyJ...' → 'eyJ...'
    const [type, token] = header?.split(' ') ?? []
    return type === 'Bearer' ? token : undefined
  }
}
roles.guard.ts — role-based access control
src/commons/guards/roles.guard.ts guard 
@Injectable()
export class RolesGuard implements CanActivate {

  canActivate(ctx: ExecutionContext): boolean {
    // Read roles allowed for this route (set via @Roles() decorator)
    const requiredRoles = this.reflector.getAllAndOverride(ROLES_KEY, [
      ctx.getHandler(),   // method-level decorator wins
      ctx.getClass(),    // class-level decorator is fallback
    ])
    if (!requiredRoles) return true   // no restriction → allow

    const { user } = ctx.switchToHttp().getRequest()
    if (!requiredRoles.includes(user?.role)) {
      throw new ForbiddenException('Insufficient role')
    }
    return true
  }
}

// Usage on a controller:
@Roles('ADMIN')
@UseGuards(AuthGuard, RolesGuard)
@Get('students')
getAllStudents() { ... }

// Or apply globally for all routes in a controller:
@UseGuards(AuthGuard, RolesGuard)
@Roles('EDUCATOR')
@Controller('lessons')
export class LessonController {}
Multi-tenant safety: The JWT payload contains orgId. Every repository method that queries the DB automatically scopes by orgId — no data from another org is ever returned, regardless of role.
08 commons/filters/ — Exception Filters
Filters catch exceptions thrown anywhere in the pipeline and convert them into a consistent JSON error shape. Two filters are layered: the specific one handles known HTTP errors, the catch-all handles everything else.
src/commons/filters/http-exception.filter.ts filter 
@Catch(HttpException)
export class HttpExceptionFilter implements ExceptionFilter {

  catch(exception: HttpException, host: ArgumentsHost) {
    const ctx        = host.switchToHttp()
    const response   = ctx.getResponse()
    const request    = ctx.getRequest()
    const statusCode = exception.getStatus()
    const exRes      = exception.getResponse()

    // Standardised error body sent to every client
    response.status(statusCode).json({
      success:   false,
      statusCode,
      requestId: request['requestId'],  // links to logs
      timestamp: new Date().toISOString(),
      path:      request.url,
      message:   typeof exRes === 'object' ? exRes['message'] : exRes,
      errors:    exRes['errors'] ?? [],   // validation errors array
    })

    this.logger.warn(`HTTP ${statusCode} on ${request.url}`, { requestId: request['requestId'] })
  }
}

// all-exception.filter.ts  — catches anything NOT HttpException
@Catch()  // empty = catch everything
export class AllExceptionFilter implements ExceptionFilter {
  catch(exception: unknown, host: ArgumentsHost) {
    this.logger.error('Unhandled exception', exception)
    // Return 500 with generic message — never leak internal details
    response.status(500).json({ success: false, message: 'Internal server error' })
  }
}
09 commons/interceptors/ — Logging & Response
Interceptors wrap the route handler execution. They see both the request (before handler) and the response (after handler). Think of them as Express middleware that can also transform what the handler returns.
src/commons/interceptors/logging.interceptor.ts interceptor 
@Injectable()
export class LoggingInterceptor implements NestInterceptor {

  intercept(ctx: ExecutionContext, next: CallHandler): Observable<any> {
    const req       = ctx.switchToHttp().getRequest()
    const startTime = Date.now()

    this.logger.log(`→ ${req.method} ${req.url}`, { requestId: req['requestId'] })

    // next.handle() executes the route handler
    // tap() runs AFTER the handler resolves (non-blocking)
    return next.handle().pipe(
      tap(() => {
        const duration = Date.now() - startTime
        this.logger.log(`← ${req.method} ${req.url} +${duration}ms`, {
          requestId: req['requestId'],
          userId:    req.user?.sub,
        })
      })
    )
  }
}
src/commons/interceptors/response.interceptor.ts interceptor 
// Wraps all successful responses in a consistent envelope
@Injectable()
export class ResponseInterceptor implements NestInterceptor {

  intercept(ctx: ExecutionContext, next: CallHandler): Observable<any> {
    const req = ctx.switchToHttp().getRequest()

    return next.handle().pipe(
      map(data => ({
        success:   true,
        statusCode: ctx.switchToHttp().getResponse().statusCode,
        requestId: req['requestId'],
        timestamp: new Date().toISOString(),
        data,              // the actual controller return value goes here
      }))
    )
  }
}

// Every API response looks like:
// { "success": true, "statusCode": 200, "requestId": "...", "data": { ... } }
10 commons/pipes/ — Validation & Transform
Pipes run on incoming data right before the route handler. The global ValidationPipe (wired in main.ts) applies automatically. Custom pipes handle specific transforms.
src/commons/pipes/validation.pipe.ts pipe 
// Global ValidationPipe config — applied in main.ts
// class-validator + class-transformer power this
new ValidationPipe({
  whitelist:           true,  // strip properties not in DTO
  transform:           true,  // auto-convert '123' → 123 for @IsInt()
  forbidNonWhitelisted:true,  // throw if unknown props sent
  exceptionFactory: (errors) => {
    // Convert class-validator errors into structured response
    const formatted = errors.map(e => ({
      field:    e.property,
      messages: Object.values(e.constraints ?? {}),
    }))
    throw new BadRequestException({ message: 'Validation failed', errors: formatted })
  }
})

// parse-int.pipe.ts — for path params that must be integers
@Injectable()
export class ParseIntPipe implements PipeTransform {
  transform(value: string) {
    const parsed = parseInt(value, 10)
    if (isNaN(parsed)) throw new BadRequestException('ID must be a number')
    return parsed
  }
}

// bcrypt helpers — used by AuthService for password hashing
export async function hashPassword(plain: string): Promise<string> {
  return bcrypt.hash(plain, 12)   // 12 rounds is the sweet spot
}
export async function verifyPassword(plain: string, hashed: string): Promise<boolean> {
  return bcrypt.compare(plain, hashed)
}

// ─── token.util.ts ───────────────────────────────────────────────────────
// Generates system passwords (10 char alphanumeric) for new accounts
export function generatePassword(length = 10): string {
  const chars = 'ABCDEFGHJKMNPQRSTUVWXYZabcdefghjkmnpqrstuvwxyz23456789'
  return Array.from({ length }, () => chars[Math.floor(Math.random() * chars.length)]).join('')
}
// Creates short unique IDs for Educator auto-IDs
export function generateEducatorId(orgId: string): string {
  return `EDU-${orgId.slice(0,4).toUpperCase()}-${nanoid(6)}`
}

// ─── date.util.ts ────────────────────────────────────────────────────────
// Week index computation (matches Section 10.3 of spec)
// Given a class start date, weekdays[], and session date → returns '1.2' style label
export function computeWeekLabel(startDate: Date, weekdays: string[], sessionDate: Date): string {
  const calendarWeek = differenceInCalendarWeeks(sessionDate, startDate) + 1
  const dayIndex     = weekdays.indexOf(format(sessionDate, 'EEE')) + 1
  if (weekdays.length === 1) return `Week ${calendarWeek}`
  return `Week ${calendarWeek}.${dayIndex}`
  // Examples: 'Week 1', 'Week 1.2', 'Week 3.1'
}

auth.repository.ts — DB queries
src/modules/auth/auth.repository.ts repository 
@Injectable()
export class AuthRepository {
  constructor(private db: DatabaseService) {}

  async findUserByEmail(email: string) {
    return this.db.user.findUnique({ where: { email } })
  }

  async saveRefreshToken(userId: string, tokenHash: string, expiresAt: Date) {
    // Rotate: revoke all previous tokens for this user first
    await this.db.refreshToken.updateMany({
      where:  { userId, revokedAt: null },
      data:   { revokedAt: new Date() },
    })
    return this.db.refreshToken.create({
      data: { userId, tokenHash, expiresAt }
    })
  }

  async findValidRefreshToken(userId: string, tokenHash: string) {
    return this.db.refreshToken.findFirst({
      where: { userId, tokenHash, revokedAt: null, expiresAt: { gt: new Date() } }
    })
  }
}
auth.service.ts — business logic
src/modules/auth/auth.service.ts service 
@Injectable()
export class AuthService {
  constructor(
    private repo:      AuthRepository,
    private jwt:       JwtService,
    private jwtConfig: JwtConfig,
  ) {}

  async login(dto: LoginDto): Promise<TokenPair> {
    // 1. Find user — same error for bad email OR bad password (timing-safe)
    const user = await this.repo.findUserByEmail(dto.email)
    if (!user) throw new UnauthorizedException('Invalid credentials')

    // 2. Compare passwords
    const valid = await verifyPassword(dto.password, user.passwordHash)
    if (!valid) throw new UnauthorizedException('Invalid credentials')

    // 3. Check account not blocked or in invalid status
    if (user.isBlocked) throw new ForbiddenException('Account is blocked')
    if (user.status === 'DROPPED' || user.status === 'GRADUATED') {
      throw new ForbiddenException('Account is inactive')
    }

    // 4. Issue tokens
    return this.issueTokenPair(user)
  }

  private async issueTokenPair(user: User): Promise<TokenPair> {
    const payload = { sub: user.id, role: user.role, orgId: user.orgId }

    const accessToken  = this.jwt.sign(payload, {
      secret:    this.jwtConfig.accessSecret,
      expiresIn: this.jwtConfig.accessExpiry,   // 15m
    })
    const refreshToken = this.jwt.sign(payload, {
      secret:    this.jwtConfig.refreshSecret,
      expiresIn: this.jwtConfig.refreshExpiry,  // 7d
    })

    // Store hashed refresh token for rotation/revocation
    await this.repo.saveRefreshToken(
      user.id,
      await hashPassword(refreshToken),
      addDays(new Date(), 7)
    )
    return { accessToken, refreshToken }
  }

  async refresh(dto: RefreshTokenDto): Promise<TokenPair> {
    let payload: JwtPayload
    try {
      payload = this.jwt.verify(dto.refreshToken, { secret: this.jwtConfig.refreshSecret })
    } catch {
      throw new UnauthorizedException('Invalid refresh token')
    }

    const tokenHash = await hashPassword(dto.refreshToken)
    const stored    = await this.repo.findValidRefreshToken(payload.sub, tokenHash)
    if (!stored) throw new UnauthorizedException('Token revoked or expired')

    const user = await this.repo.findUserByEmail(payload.sub)
    return this.issueTokenPair(user)   // issues new pair + revokes old token
  }
}

14 Request Pipeline — How a Request Flows
Every HTTP request passes through these layers in order. Understanding this flow is essential for knowing where to put logic.
RequestIdMiddleware
→
Helmet · CORS · Compression
→
AuthGuard (JWT)
→
RolesGuard
→
LoggingInterceptor
→
ResponseInterceptor
→
ValidationPipe
→
Route Handler
→
HttpExceptionFilter
On happy path

Request enters → RequestId stamped → JWT verified → Role checked → interceptors run before → ValidationPipe validates body → Handler executes → ResponseInterceptor wraps result → Client gets { success: true, data: ... }

On error path

Any layer throws → exception bubbles up → HttpExceptionFilter catches known errors → AllExceptionFilter catches unknown errors → Client gets { success: false, message: ..., requestId: ... }. RequestId lets you find the full trace in Winston logs.

Multi-tenant enforcement point: After AuthGuard runs, req.user.orgId is always set. The @CurrentUser('orgId') decorator extracts it. Every repository method receives orgId and uses it in WHERE org_id = $orgId. This is the single enforcement boundary — all data isolation flows from here.

// Attempt control: one active attempt per student (resume across tabs/devices)
// Score publishing: per-student or all-at-once, unpublishable
// On grade lock: auto-publish all unpublished scores

// ─── modules/attendance/ ─────────────────────────────────────────────────
// Per session (not per calendar day). Skips calendar event days automatically.
// Auto-mark Present from assessment submissions (Section 15.2)
// Weekly view: Week 1, Week 1.1/1.2 etc. (see date.util.ts)

// ─── modules/grade/ ──────────────────────────────────────────────────────
// Per term within semester. Rubric weights must total 100%.
// Rubric locks on first enrollment. Default view vs Clean view.
// Grade lock: permanent unless Admin overrides. Auto-lock if deadline missed.
// Emits: 'grade.locked' → auto-publishes all scores → notifies students

// ─── modules/grading-scale/ ──────────────────────────────────────────────
// Admin configures per level section. Must cover 0-100 fully, no gaps/overlaps.
// Locks when first grade is locked for that level section in school year.
// Unlocks at start of new school year.

// ─── modules/calendar/ ───────────────────────────────────────────────────
// Admin manages org-wide calendar events per school year.
// Holiday/No Class Day: attendance sessions auto-skipped
// Exam Week / Special Event: informational only
// Emits: 'calendar.event.created' → AttendanceService, MeetingService

// ─── modules/meeting/ ────────────────────────────────────────────────────
// Built-in video room. No recording. Educator controls everything.
// Suppressed on calendar event days.
// Features: lesson presentation mode (real-time forward/back for all)
// Non-invited students can request join → educator accepts/declines

// ─── modules/notification/ ───────────────────────────────────────────────
// In-app only. Simple list. Listens to domain events via EventBus.
// Retention: 90 days → auto-archive (cron job)
// No email/SMS integration needed per spec

// ─── modules/audit-log/ ──────────────────────────────────────────────────
// Two tiers: AdminAuditLog (high-impact) + EducatorActivityLog (class-level)
// Never deleted. Searchable by Admin.
// Populated by EventBus listeners — not called directly by services

// ─── modules/export/ ─────────────────────────────────────────────────────
// PDF class cards (per student) + CSV full class export
// Both Admin and Educator can trigger for their scope
// Uses: pdfmake or puppeteer for PDF generation

// ─── modules/analytics/ ──────────────────────────────────────────────────
// Admin dashboard only. Aggregate views — no live class internals.
// Enrollment counts, pending students, grade distribution (post-lock)
Org isolation rule: Every repository in every module must include WHERE org_id = $orgId in all queries. The orgId always comes from the JWT payload via @CurrentUser('orgId') — never from the request body or query params.
