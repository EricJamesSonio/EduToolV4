## main.ts Bootstrap

```typescript
async function bootstrap() {
  const app = await NestFactory.create(AppModule);

  app.use(helmet());

  app.enableCors({
    origin: process.env.FRONTEND_URL,
    credentials: true,
  });

  app.setGlobalPrefix('api/v1');

  app.useGlobalPipes(
    new ValidationPipe({
      whitelist: true,            // strips unknown fields
      forbidNonWhitelisted: true, // throws if unknown fields sent
      transform: true,
    }),
  );

  app.useGlobalFilters(new HttpExceptionFilter());
  app.useGlobalInterceptors(new LoggingInterceptor(), new ResponseInterceptor());

  // Swagger — dev/staging only
  if (process.env.NODE_ENV !== 'production') {
    const config = new DocumentBuilder()
      .setTitle('EduTool API')
      .setVersion('1.0')
      .addBearerAuth()
      .build();
    SwaggerModule.setup('api/docs', app, SwaggerModule.createDocument(app, config));
  }

  await app.listen(process.env.PORT || 3001);
}
```

## Authentication

### JWT Strategy

```typescript
// auth/strategies/jwt.strategy.ts
@Injectable()
export class JwtStrategy extends PassportStrategy(Strategy) {
  constructor(config: ConfigService) {
    super({
      jwtFromRequest: ExtractJwt.fromAuthHeaderAsBearerToken(),
      secretOrKey: config.get('JWT_SECRET'),
    });
  }

  async validate(payload: { sub: string; orgId: string; role: string }) {
    return {
      userId: payload.sub,
      orgId: payload.orgId,
      role: payload.role,
    };
  }
}
```

### JWT Guard (apply to every protected route)

```typescript
// common/guards/jwt.guard.ts
@Injectable()
export class JwtAuthGuard extends AuthGuard('jwt') {}

// Usage on controller
@UseGuards(JwtAuthGuard)
@Get('students')
getStudents() {}
```

### Token Invalidation on Password Reset

JWTs are stateless — you need Redis to force-expire them when Admin resets a password.

```typescript
// On login — store jti in Redis
await redis.set(`token:${jti}`, userId, 'EX', 60 * 60 * 24 * 7);

// In JWT validate() — check it's still alive
const alive = await redis.exists(`token:${jti}`);
if (!alive) throw new UnauthorizedException();

// On password reset — delete all tokens for this user
const keys = await redis.keys(`token:user:${userId}:*`);
if (keys.length) await redis.del(...keys);
```

## Authorization — Roles & Org Isolation

### Roles Enum

```typescript
// common/constants/roles.enum.ts
export enum Role {
  PLATFORM_OWNER = 'PLATFORM_OWNER',
  ADMIN = 'ADMIN',
  EDUCATOR = 'EDUCATOR',
  STUDENT = 'STUDENT',
}
```

### @Roles() Decorator + Guard

```typescript
// common/decorators/roles.decorator.ts
export const Roles = (...roles: Role[]) => SetMetadata('roles', roles);

// common/guards/roles.guard.ts
@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private reflector: Reflector) {}
  canActivate(ctx: ExecutionContext): boolean {
    const required = this.reflector.get<Role[]>('roles', ctx.getHandler());
    if (!required) return true;
    const { user } = ctx.switchToHttp().getRequest();
    return required.includes(user.role);
  }
}

// Usage
@Get('students')
@Roles(Role.ADMIN)
@UseGuards(JwtAuthGuard, RolesGuard)
getStudents() {}
```

### @CurrentUser() Decorator

```typescript
// common/decorators/current-user.decorator.ts
export const CurrentUser = createParamDecorator(
  (_, ctx: ExecutionContext) => ctx.switchToHttp().getRequest().user,
);

// Usage in controller
@Get('me')
getMe(@CurrentUser() user: JwtUser) {
  return user;
}
```

### Org Isolation — The Most Important Rule

Every single DB query must be scoped to `orgId`. No exceptions.

```typescript
// WRONG
findAll() {
  return this.prisma.student.findMany();
}

// CORRECT — always pass orgId from the JWT
findAll(orgId: string) {
  return this.prisma.student.findMany({ where: { orgId } });
}
```

Pass `orgId` from the controller → service → repository on every call.

## Validation — DTOs

Every request body needs a DTO with decorators. The global `ValidationPipe` runs them automatically.

```typescript
// dto/create-student.dto.ts
export class CreateStudentDto {
  @IsString()
  @MaxLength(100)
  fullName: string;

  @IsEmail()
  email: string;

  @IsString()
  studentId: string;

  @IsEnum(LevelSection)
  levelSection: LevelSection;

  @IsString()
  gradeYear: string;

  @IsOptional()
  @IsUUID()
  sectionId?: string;
}
```

Rules:
- Every field must have at least one validator
- `@IsOptional()` must still have a type validator after it
- Never use `any` in DTOs
- Never return raw Prisma objects — always map to a response DTO

## Error Handling

### Global Exception Filter

```typescript
// common/filters/http-exception.filter.ts
@Catch()
export class HttpExceptionFilter implements ExceptionFilter {
  catch(exception: unknown, host: ArgumentsHost) {
    const ctx = host.switchToHttp();
    const res = ctx.getResponse();
    const req = ctx.getRequest();

    let status = 500;
    let message = 'Internal server error';
    let code = 'INTERNAL_ERROR';

    if (exception instanceof HttpException) {
      status = exception.getStatus();
      const body = exception.getResponse() as any;
      message = body.message ?? body;
      code = body.code ?? 'HTTP_ERROR';
    } else if (exception instanceof Prisma.PrismaClientKnownRequestError) {
      if (exception.code === 'P2002') {
        status = 409;
        message = 'A record with this value already exists.';
        code = 'DUPLICATE_RECORD';
      }
      if (exception.code === 'P2025') {
        status = 404;
        message = 'Record not found.';
        code = 'NOT_FOUND';
      }
    }

    if (status >= 500) {
      console.error(exception);
    }

    res.status(status).json({
      success: false,
      statusCode: status,
      code,
      message: Array.isArray(message) ? message : [message],
      path: req.url,
      timestamp: new Date().toISOString(),
    });
  }
}
```

### Named Business Exceptions

Never throw raw strings. One class per business rule so the frontend can `switch` on `code`.

### Consistent Response Shape

Every API response — success or error — looks the same:

```json
// Success
{
  "success": true,
  "statusCode": 200,
  "data": { ... }
}

// Paginated success
{
  "success": true,
  "statusCode": 200,
  "data": [ ... ],
  "meta": { "total": 150, "page": 1, "limit": 20, "totalPages": 8 }
}

// Error
{
  "success": false,
  "statusCode": 422,
  "code": "SECTION_CAPACITY_FULL",
  "message": ["Section 'Block A' is full (30 students)."],
  "path": "/api/v1/students",
  "timestamp": "2025-01-13T10:00:00.000Z"
}
```

### Response Interceptor (wraps all success responses)

```typescript
// common/interceptors/response.interceptor.ts
@Injectable()
export class ResponseInterceptor implements NestInterceptor {
  intercept(ctx: ExecutionContext, next: CallHandler): Observable<any> {
    return next.handle().pipe(
      map((data) => ({
        success: true,
        statusCode: ctx.switchToHttp().getResponse().statusCode,
        ...(data?.meta ? { data: data.data, meta: data.meta } : { data }),
      })),
    );
  }
}
```

## Logging

### Request Logging Interceptor

```typescript
// common/interceptors/logging.interceptor.ts
@Injectable()
export class LoggingInterceptor implements NestInterceptor {
  private logger = new Logger('HTTP');

  intercept(ctx: ExecutionContext, next: CallHandler): Observable<any> {
    const req = ctx.switchToHttp().getRequest();
    const { method, url } = req;
    const start = Date.now();

    return next.handle().pipe(
      tap(() => this.logger.log(`${method} ${url} — ${Date.now() - start}ms`)),
    );
  }
}
```

### What to Log / Not Log

```
✅ Log:   requests (method, path, duration), job start/complete/fail,
          5xx errors with stack, auth failures
❌ Never: passwords, JWT tokens, full request bodies, student PII
```

## Prisma — Rules

### Soft Delete Middleware

Intercept all deletes on soft-delete models so hard deletes are impossible.

```typescript
// prisma.service.ts
this.$use(async (params, next) => {
  const softDeleteModels = ['Class', 'Assessment', 'Lesson', 'Enrollment', 'Meeting'];

  if (softDeleteModels.includes(params.model)) {
    if (params.action === 'findMany' || params.action === 'findFirst') {
      params.args.where = { ...params.args.where, deletedAt: null };
    }
    if (params.action === 'delete') {
      params.action = 'update';
      params.args.data = { deletedAt: new Date() };
    }
  }
  return next(params);
});
```

### Transactions

Use for any write that touches 2+ tables.

```typescript
await this.prisma.$transaction(async (tx) => {
  const enrollment = await tx.enrollment.create({ data: { ... } });
  await tx.adminAuditLog.create({ data: { ... } });
  return enrollment;
});
```

### Repository Pattern

```typescript
// student.repository.ts
@Injectable()
export class StudentRepository {
  constructor(private prisma: PrismaService) {}

  findAll(orgId: string, page: number, limit: number) {
    return this.prisma.$transaction([
      this.prisma.student.findMany({
        where: { orgId },
        skip: (page - 1) * limit,
        take: limit,
      }),
      this.prisma.student.count({ where: { orgId } }),
    ]);
  }

  findById(id: string, orgId: string) {
    return this.prisma.student.findFirst({ where: { id, orgId } });
  }
}
```

## Security Checklist (before going live)

```
Auth
  [ ] JwtAuthGuard on every protected route
  [ ] Token invalidation working on password reset
  [ ] bcrypt rounds = 12

Authorization
  [ ] @Roles() on every endpoint
  [ ] orgId in every WHERE clause
  [ ] Students cannot access other students' data
  [ ] Educators only see their own classes

Input
  [ ] ValidationPipe whitelist: true globally
  [ ] Request size limits set (10mb for import, 1mb default)

Sensitive Data
  [ ] Passwords never in logs or API responses
  [ ] generated_password only exposed in credential export endpoint
  [ ] JWT tokens never logged

Transport
  [ ] HTTPS + HSTS in production
  [ ] CORS locked to frontend URL only
  [ ] helmet() applied

Rate Limiting
  [ ] Login: 5/min
  [ ] General: 100/min
  [ ] Bulk import: 3/min
```
